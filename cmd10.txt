6. ADD NEW RUN-FROM-PACKAGE DEPLOYMENT
============================================================

Add a new deployment step after Terraform infrastructure deployment and
after the Function ZIP package has been created.

Use:

- name: Deploy Function package
  if: ${{ inputs.enable_file_processing == 'YES' }}
  shell: bash
  run: |
    set -euo pipefail

    STORAGE_ACCOUNT="vtxcusvdpdataplatfe1a54"
    CONTAINER="function-packages"
    FUNCTION_APP="sox-dev-function"
    RESOURCE_GROUP="cpe-centralus-corptoolsdev01-a-vdp-dataplatform-gov-service-development-rg"
    PACKAGE="${GITHUB_WORKSPACE}/azure-infra/function/sox-function.zip"
    BLOB_NAME="sox-function-${GITHUB_SHA}.zip"

    echo "Validating Function package..."

    if [ ! -f "$PACKAGE" ]; then
      echo "ERROR: Function package not found: $PACKAGE"
      exit 1
    fi

    echo "Uploading Function package..."

    az storage blob upload \
      --account-name "$STORAGE_ACCOUNT" \
      --container-name "$CONTAINER" \
      --name "$BLOB_NAME" \
      --file "$PACKAGE" \
      --auth-mode login \
      --overwrite \
      --output none

    PACKAGE_URL="https://${STORAGE_ACCOUNT}.blob.core.windows.net/${CONTAINER}/${BLOB_NAME}"

    echo "Configuring WEBSITE_RUN_FROM_PACKAGE..."

    az functionapp config appsettings set \
      --name "$FUNCTION_APP" \
      --resource-group "$RESOURCE_GROUP" \
      --settings "WEBSITE_RUN_FROM_PACKAGE=${PACKAGE_URL}" \
      --output none

    echo "Restarting Function App..."

    az functionapp restart \
      --name "$FUNCTION_APP" \
      --resource-group "$RESOURCE_GROUP"

    echo "Synchronizing Function triggers..."

    az rest \
      --method post \
      --url "https://management.azure.com/subscriptions/${ARM_SUBSCRIPTION_ID}/resourceGroups/${RESOURCE_GROUP}/providers/Microsoft.Web/sites/${FUNCTION_APP}/syncfunctiontriggers?api-version=2022-03-01"

    echo "Function package deployment completed successfully."
